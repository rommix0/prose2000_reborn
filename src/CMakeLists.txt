# C decompilation of the Prose 2000 firmware.
#
#   cmake -S src -B <dir> -G Ninja [-DPROSE_ROM_DIR=<abs path to prose_v3>] && cmake --build <dir>
#
# PROSE_ROM_DIR also generates the instruction-level DSP translation (tools/dsp_recompile.py) into the build
# directory. That file embeds the DSP program ROM, so it is never written into the source tree.
cmake_minimum_required(VERSION 3.20)
project(prose_decomp LANGUAGES C)
# Windows: 64-bit by default, since a 64-bit program (the usual Python among them) cannot load a 32-bit DLL. The
# compiler decides: a 64-bit MinGW-w64 gcc (x86_64-w64-mingw32) builds 64-bit code, a 32-bit one (i686) 32-bit code.
option(PROSE_32BIT "Windows: build 32-bit, for 32-bit programs, with a 32-bit MinGW-w64 gcc" OFF)
if(WIN32)
	math(EXPR prose_bits "${CMAKE_SIZEOF_VOID_P} * 8")
	if(prose_bits EQUAL 32 AND NOT PROSE_32BIT)
		message(FATAL_ERROR "${CMAKE_C_COMPILER} builds 32-bit code, and the Windows build is 64-bit by default: put a "
			"64-bit MinGW-w64 gcc (x86_64-w64-mingw32) first on PATH, or add -DPROSE_32BIT=ON for a 32-bit build "
			"(README). Delete CMakeCache.txt in the build folder before configuring again with another compiler.")
	elseif(prose_bits EQUAL 64 AND PROSE_32BIT)
		message(FATAL_ERROR "PROSE_32BIT is on, but ${CMAKE_C_COMPILER} builds 64-bit code: use a 32-bit MinGW-w64 "
			"gcc (i686-w64-mingw32).")
	endif()
	message(STATUS "Building ${prose_bits}-bit")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
# the libraries also go into the shared library (prose.dll / libprose.so)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
	add_compile_options(-Wall -Wextra)
endif()

add_library(prose_wave STATIC common/prose_wave.c)
target_include_directories(prose_wave PUBLIC common)
if(WIN32)
	target_link_libraries(prose_wave PUBLIC winmm)
endif()

# The ROM's data contents (generated by tools/rom_extract.py) and the ROM image built from them.
add_library(prose_data STATIC data/prose_data.c)
target_include_directories(prose_data PUBLIC data)
add_library(prose_common STATIC common/prose_rom.c)
target_include_directories(prose_common PUBLIC common)
target_link_libraries(prose_common PUBLIC prose_data)

add_library(prose_synth STATIC dsp/prose_synth.c dsp/dsp_rom.c)
target_include_directories(prose_synth PUBLIC dsp)
target_link_libraries(prose_synth PUBLIC prose_data)

add_library(prose_frame STATIC frame/frame_build.c)
target_include_directories(prose_frame PUBLIC frame)
target_link_libraries(prose_frame PUBLIC prose_common)

add_library(prose_paramgen STATIC paramgen/pg_mem.c paramgen/pg_nodes.c paramgen/pg_run.c paramgen/pg_segment.c
	paramgen/pg_tracks.c paramgen/pg_loci.c paramgen/pg_context.c)
target_include_directories(prose_paramgen PUBLIC paramgen)
target_link_libraries(prose_paramgen PUBLIC prose_common)

add_library(prose_prosody STATIC prosody/pr_stage.c prosody/pr_phrase.c prosody/pr_duration.c prosody/pr_f0.c)
target_include_directories(prose_prosody PUBLIC prosody)
target_link_libraries(prose_prosody PUBLIC prose_paramgen)

add_library(prose_lexical STATIC lexical/lx_stage.c lexical/lx_lexicon.c lexical/lx_lts.c lexical/lx_allophone.c)
target_include_directories(prose_lexical PUBLIC lexical)
target_link_libraries(prose_lexical PUBLIC prose_prosody)
add_library(prose_input STATIC input/in_stage.c input/in_host.c input/in_escape.c input/in_reset.c input/in_boot.c
	input/in_dsp.c)
target_include_directories(prose_input PUBLIC input)
target_link_libraries(prose_input PUBLIC prose_prosody prose_frame)
add_library(prose_textrules STATIC textrules/tr_stage.c textrules/tr_phoneme.c)
target_include_directories(prose_textrules PUBLIC textrules)
target_link_libraries(prose_textrules PUBLIC prose_paramgen)

# v1.1 (1983, REFERENCE §16), and its frames on the v3.12 DSP model (its own DSP program is lost)
add_library(prose_v1_data STATIC data/prose_v1_data.c)
target_include_directories(prose_v1_data PUBLIC data)
add_library(prose_v1 STATIC v1/v1_mem.c v1/v1_board.c v1/v1_host.c v1/v1_escape.c v1/v1_nodes.c v1/v1_input.c
	v1/v1_loop.c v1/v1_textrules.c v1/v1_lexical.c v1/v1_prosody.c v1/v1_paramgen.c v1/v1_frame.c)
target_include_directories(prose_v1 PUBLIC v1)
target_link_libraries(prose_v1 PUBLIC prose_v1_data)
add_library(prose_v1_map STATIC v1/v1_map.c)
target_include_directories(prose_v1_map PUBLIC v1)
target_link_libraries(prose_v1_map PUBLIC prose_frame)

# The speech library (API.md): both firmware versions behind the prose_* API, and a command-line speaker.
add_library(prose SHARED dll/prose_api.c dll/prose_engine.c dll/prose_eng3.c dll/prose_eng1.c dll/prose_params.c
	dll/prose_sys.c)
target_include_directories(prose PUBLIC include PRIVATE dll)
target_compile_definitions(prose PRIVATE PROSE_BUILD_DLL)
target_link_libraries(prose PRIVATE prose_input prose_textrules prose_lexical prose_prosody prose_paramgen prose_frame
	prose_synth prose_wave prose_v1 prose_v1_map)
set_target_properties(prose PROPERTIES C_VISIBILITY_PRESET hidden PREFIX "" IMPORT_PREFIX "")
if(MINGW) # no libgcc DLL to ship
	target_link_options(prose PRIVATE -static-libgcc)
endif()
if(NOT WIN32)
	set_target_properties(prose PROPERTIES PREFIX "lib")
	target_link_libraries(prose PRIVATE ${CMAKE_DL_LIBS} pthread)
endif()
add_executable(prose_say cli/prose_say.c)
target_link_libraries(prose_say PRIVATE prose)
# Linux: the programs also find libprose.so in their own folder (Windows looks next to the .exe anyway)
set_target_properties(prose_say PROPERTIES BUILD_RPATH "$ORIGIN")

# The same library linked statically, for tools that also use the firmware's internal hooks: pitch_trace writes the
# F0 targets and their parts (PITCH_SYSTEM.md; pitch_graphs/ plots them)
add_library(prose_static STATIC dll/prose_api.c dll/prose_engine.c dll/prose_eng3.c dll/prose_eng1.c
	dll/prose_params.c dll/prose_sys.c)
target_include_directories(prose_static PUBLIC include PRIVATE dll)
target_compile_definitions(prose_static PUBLIC PROSE_STATIC)
target_link_libraries(prose_static PUBLIC prose_input prose_textrules prose_lexical prose_prosody prose_paramgen
	prose_frame prose_synth prose_wave prose_v1 prose_v1_map)
if(NOT WIN32)
	target_link_libraries(prose_static PUBLIC ${CMAKE_DL_LIBS} pthread)
endif()
add_executable(pitch_trace cli/pitch_trace.c)
target_link_libraries(pitch_trace PRIVATE prose_static)

# The examples of API.md (../samples)
option(PROSE_SAMPLES "Build the samples" ON)
if(PROSE_SAMPLES)
	foreach(sample speak_events speak_buffer raw_glide params_export params_import glottal_pulse lip_sync)
		add_executable(${sample} ../samples/${sample}.c)
		target_link_libraries(${sample} PRIVATE prose)
		set_target_properties(${sample} PROPERTIES BUILD_RPATH "$ORIGIN")
		if(NOT WIN32)
			target_link_libraries(${sample} PRIVATE m)
		endif()
	endforeach()
endif()

# Which firmware to build: 3 = v3.4.1 (default), 1 = the older v1.1 (1983, REFERENCE §16).
set(PROSE_VERSION 3 CACHE STRING "Firmware version to build: 3 (v3.4.1) or 1 (v1.1)")
set_property(CACHE PROSE_VERSION PROPERTY STRINGS 3 1)

if(PROSE_VERSION EQUAL 1)
	add_executable(v1_replay tests/v1_replay.c)
	target_link_libraries(v1_replay PRIVATE prose_v1)
	add_executable(v1_boot_check tests/v1_boot_check.c)
	target_link_libraries(v1_boot_check PRIVATE prose_v1)
	add_executable(v1_pipeline_play tests/v1_pipeline_play.c)
	target_link_libraries(v1_pipeline_play PRIVATE prose_v1 prose_v1_map prose_synth prose_wave)
	return()
endif()

add_executable(chain_replay tests/chain_replay.c)

set(PROSE_ROM_DIR "" CACHE PATH "Directory with the v3.12 DSP ROM dumps (enables the instruction-level translation)")
if(PROSE_ROM_DIR)
	find_package(Python3 REQUIRED COMPONENTS Interpreter)
	set(rec_c ${CMAKE_CURRENT_BINARY_DIR}/dsp_rec.c)
	add_custom_command(OUTPUT ${rec_c}
		COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/../tools/dsp_recompile.py ${PROSE_ROM_DIR} -o ${rec_c}
		DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../tools/dsp_recompile.py ${CMAKE_CURRENT_SOURCE_DIR}/../tools/dsp_disasm.py
		COMMENT "Translating the DSP program to C")
	add_library(dsp_rec STATIC ${rec_c})
	target_include_directories(dsp_rec PUBLIC dsp)
	target_compile_options(dsp_rec PRIVATE -Wno-unused-function -Wno-unused-label)
	target_link_libraries(dsp_replay PRIVATE dsp_rec)
	target_compile_definitions(dsp_replay PRIVATE HAVE_DSP_REC)
endif()
